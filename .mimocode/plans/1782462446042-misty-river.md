@router.patch("/users/{user_id}")
@require_role(UserRole.SUPER_ADMIN)
async def update_user_status(
    user_id: str,
    body: UpdateUserStatusRequest,
    current_user: UserModel = Depends(get_current_rbac_user),
):
    """Enable or disable a user. Requires super_admin role."""
    if user_id == current_user.id:
        raise HTTPException(status_code=400, detail="Cannot change your own status")

    sf = get_session_factory()
    if sf is None:
        raise HTTPException(status_code=500, detail="Database not initialized")

    async with sf() as session:
        stmt = select(UserModel).where(UserModel.id == user_id)
        try:
            stmt = stmt.with_for_update()
        except Exception:
            pass
        result = await session.execute(stmt)
        user = result.scalar_one_or_none()
        if user is None:
            raise HTTPException(status_code=404, detail="User not found")

        # Prevent disabling the last active super_admin
        if body.disabled and user.role == UserRole.SUPER_ADMIN:
            count_stmt = select(func.count()).select_from(UserModel).where(
                UserModel.role == UserRole.SUPER_ADMIN,
                UserModel.disabled.is_not(True),
            )
            try:
                count_stmt = count_stmt.with_for_update()
            except Exception:
                pass
            active_count = (await session.execute(count_stmt)).scalar() or 0
            if active_count <= 1:
                raise HTTPException(status_code=400, detail="Cannot disable the last active super_admin")

        user.disabled = body.disabled
        await session.commit()

    action = "disabled" if body.disabled else "enabled"
    logger.warning("User %s: user_id=%s, by=%s", action, user_id, current_user.id)
    return {"success": True, "disabled": body.disabled}
```

@pytest.mark.asyncio
async def test_can_disable_and_enable_user(self, super_admin_cookies):
    """超级管理员可以禁用和启用用户"""
    csrf = super_admin_cookies.get("csrf_token", "")
    headers = {"X-CSRF-Token": csrf}

    # 找一个可操作的普通用户
    async with httpx.AsyncClient() as client:
        users_resp = await client.get(
            f"{BASE_URL}/api/admin/users",
            cookies=super_admin_cookies,
        )
        target = None
        for u in users_resp.json().get("users", []):
            if u.get("role") == "user" and not u.get("disabled"):
                target = u
                break
        if not target:
            pytest.skip("No eligible user to test disable/enable")

        user_id = target["id"]

        # 禁用
        disable_resp = await client.patch(
            f"{BASE_URL}/api/admin/users/{user_id}",
            cookies=super_admin_cookies,
            headers=headers,
            json={"disabled": True},
        )
        assert disable_resp.status_code == 200
        assert disable_resp.json().get("disabled") is True

        # 启用
        enable_resp = await client.patch(
            f"{BASE_URL}/api/admin/users/{user_id}",
            cookies=super_admin_cookies,
            headers=headers,
            json={"disabled": False},
        )
        assert enable_resp.status_code == 200
        assert enable_resp.json().get("disabled") is False
```

@pytest.mark.asyncio
async def test_dept_admin_cannot_disable_user(self, dept_admin_cookies):
    """部门管理员不能禁用用户"""
    csrf = dept_admin_cookies.get("csrf_token", "")
    async with httpx.AsyncClient() as client:
        resp = await client.patch(
            f"{BASE_URL}/api/admin/users/some-id",

            headers={"X-CSRF-Token": csrf},
            json={"disabled": True},
        )
